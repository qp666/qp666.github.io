    sonRef.value.fatherSon(new Date().getTime());
};

    sonMsg.value = (val);

vuex 的使用

 //store的index.js中
 import { createStore } from "vuex";

export default createStore({
  state: {
    num: 0,
    num2: 0,
  },
  getters: {
    resetNum: (state) => {
      return state.num * 5;
    }
  },

  mutations: {
    increment(state, val) {
      state.num += val;
    },
    testAsync(state, val) {
      new Promise((resolve, reject) => {
        setTimeout(() => {
          resolve(val)
        }, 1000)
      }).then((res) => {
        state.num += res;
      })
    },
    test(state,val) {
      state.num2 += val;
    }
  },
  actions: {
    testAsync2(context,val) {
      new Promise((resolve, reject) => {
        setTimeout(() => {
          resolve()
        }, 1000)
      }).then((res) => {
        context.commit("test",val)
      })
    }
  },
  modules: {},
});

    store.commit("testAsync", 2);
